cmake_minimum_required(VERSION 3.20)
project(mini_vhdl_frontend LANGUAGES CXX)

include(ExternalProject)

find_package(Python3 REQUIRED COMPONENTS Interpreter)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

set(HDLCONVERTOR_PREFIX "${CMAKE_BINARY_DIR}/_deps/hdlConvertor")
set(HDLCONVERTOR_SOURCE_DIR "${HDLCONVERTOR_PREFIX}/src/hdlConvertor")
set(HDLCONVERTOR_BINARY_DIR "${HDLCONVERTOR_PREFIX}/build")
set(HDLCONVERTOR_VENV_DIR "${HDLCONVERTOR_PREFIX}/venv")
set(HDLCONVERTOR_LIBRARY
    "${HDLCONVERTOR_BINARY_DIR}/src/${CMAKE_STATIC_LIBRARY_PREFIX}hdlConvertor_cpp_static${CMAKE_STATIC_LIBRARY_SUFFIX}")
set(HDLCONVERTOR_CORE_LIBRARY
    "${HDLCONVERTOR_BINARY_DIR}/src/${CMAKE_STATIC_LIBRARY_PREFIX}hdlConvertor_core_static${CMAKE_STATIC_LIBRARY_SUFFIX}")
set(HDLCONVERTOR_VHDL_LIBRARY
    "${HDLCONVERTOR_BINARY_DIR}/${CMAKE_STATIC_LIBRARY_PREFIX}vhdl${CMAKE_STATIC_LIBRARY_SUFFIX}")
set(HDLCONVERTOR_SV_LIBRARY
    "${HDLCONVERTOR_BINARY_DIR}/${CMAKE_STATIC_LIBRARY_PREFIX}sv2017${CMAKE_STATIC_LIBRARY_SUFFIX}")
set(HDLCONVERTOR_VERILOG_PREPROC_LIBRARY
    "${HDLCONVERTOR_BINARY_DIR}/${CMAKE_STATIC_LIBRARY_PREFIX}verilogPreproc${CMAKE_STATIC_LIBRARY_SUFFIX}")
set(HDLCONVERTOR_ANTLR4_LIBRARY
    "${HDLCONVERTOR_BINARY_DIR}/subprojects/antlr4-runtime/${CMAKE_STATIC_LIBRARY_PREFIX}antlr4_static${CMAKE_STATIC_LIBRARY_SUFFIX}")

if(CMAKE_CONFIGURATION_TYPES)
    set(HDLCONVERTOR_BUILD_TYPE "$<CONFIG>")
elseif(CMAKE_BUILD_TYPE)
    set(HDLCONVERTOR_BUILD_TYPE "${CMAKE_BUILD_TYPE}")
else()
    set(HDLCONVERTOR_BUILD_TYPE "Debug")
endif()

file(MAKE_DIRECTORY "${HDLCONVERTOR_SOURCE_DIR}/include")
file(MAKE_DIRECTORY
    "${HDLCONVERTOR_SOURCE_DIR}/subprojects/antlr4-runtime/runtime/Cpp/runtime/src")
file(MAKE_DIRECTORY "${HDLCONVERTOR_BINARY_DIR}/include")

ExternalProject_Add(hdlConvertor_external
    PREFIX "${HDLCONVERTOR_PREFIX}"
    GIT_REPOSITORY https://github.com/Nic30/hdlConvertor.git
    GIT_TAG 03c081a307850dc3c438df36592fc67b1ef6cfc0
    GIT_SHALLOW TRUE
    GIT_SUBMODULES ""
    GIT_SUBMODULES_RECURSE FALSE
    UPDATE_DISCONNECTED TRUE
    SOURCE_DIR "${HDLCONVERTOR_SOURCE_DIR}"
    BINARY_DIR "${HDLCONVERTOR_BINARY_DIR}"
    CONFIGURE_COMMAND
        "${CMAKE_COMMAND}"
        -DHOST_PYTHON=${Python3_EXECUTABLE}
        -DHDLCONVERTOR_SOURCE_DIR=${HDLCONVERTOR_SOURCE_DIR}
        -DHDLCONVERTOR_BINARY_DIR=${HDLCONVERTOR_BINARY_DIR}
        -DHDLCONVERTOR_VENV_DIR=${HDLCONVERTOR_VENV_DIR}
        -DHDLCONVERTOR_BUILD_TYPE=${HDLCONVERTOR_BUILD_TYPE}
        -DHDLCONVERTOR_PYTHON_PACKAGE=ON
        -DMODE=configure
        -P "${CMAKE_CURRENT_SOURCE_DIR}/cmake/BuildHdlConvertor.cmake"
    BUILD_COMMAND
        "${CMAKE_COMMAND}"
        -DHOST_PYTHON=${Python3_EXECUTABLE}
        -DHDLCONVERTOR_SOURCE_DIR=${HDLCONVERTOR_SOURCE_DIR}
        -DHDLCONVERTOR_BINARY_DIR=${HDLCONVERTOR_BINARY_DIR}
        -DHDLCONVERTOR_VENV_DIR=${HDLCONVERTOR_VENV_DIR}
        -DHDLCONVERTOR_BUILD_TYPE=${HDLCONVERTOR_BUILD_TYPE}
        -DHDLCONVERTOR_PYTHON_PACKAGE=ON
        -DMODE=build
        -P "${CMAKE_CURRENT_SOURCE_DIR}/cmake/BuildHdlConvertor.cmake"
    INSTALL_COMMAND ""
    BUILD_BYPRODUCTS
        "${HDLCONVERTOR_LIBRARY}"
        "${HDLCONVERTOR_CORE_LIBRARY}"
        "${HDLCONVERTOR_VHDL_LIBRARY}"
        "${HDLCONVERTOR_SV_LIBRARY}"
        "${HDLCONVERTOR_VERILOG_PREPROC_LIBRARY}"
        "${HDLCONVERTOR_ANTLR4_LIBRARY}"
)

add_library(hdlConvertor::hdlConvertor STATIC IMPORTED GLOBAL)
set_target_properties(hdlConvertor::hdlConvertor PROPERTIES
    IMPORTED_LOCATION "${HDLCONVERTOR_LIBRARY}"
    INTERFACE_INCLUDE_DIRECTORIES
        "${HDLCONVERTOR_SOURCE_DIR}/include;${HDLCONVERTOR_BINARY_DIR}/include;${HDLCONVERTOR_SOURCE_DIR}/subprojects/antlr4-runtime/runtime/Cpp/runtime/src"
    INTERFACE_LINK_LIBRARIES
        "${HDLCONVERTOR_SV_LIBRARY};${HDLCONVERTOR_VERILOG_PREPROC_LIBRARY};${HDLCONVERTOR_VHDL_LIBRARY};${HDLCONVERTOR_CORE_LIBRARY};${HDLCONVERTOR_ANTLR4_LIBRARY}"
)
add_dependencies(hdlConvertor::hdlConvertor hdlConvertor_external)

add_executable(vhdl_fe
    src/ast_printer.cpp
    src/main.cpp
)

target_include_directories(vhdl_fe PRIVATE src)
target_link_libraries(vhdl_fe PRIVATE hdlConvertor::hdlConvertor)
